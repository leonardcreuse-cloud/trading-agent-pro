import requests
from datetime import datetime, timedelta
from typing import Dict, List

class AnalystConsensus:
    """
    Analyst Consensus & Ratings
    - Tracks analyst ratings, price targets, revisions
    - Source: Yahoo Finance (Tier 2), Seeking Alpha (Tier 2)
    """
    
    def __init__(self):
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) trading-agent-pro'
        })
    
    def get_analyst_ratings(self, ticker: str) -> Dict:
        """
        Get consensus analyst ratings
        - Returns: Buy/Hold/Sell counts and percentages
        
        Source: Yahoo Finance (Tier 2)
        Returns: {ticker, buy_count, hold_count, sell_count, consensus_rating, source}
        """
        
        try:
            url = f"https://finance.yahoo.com/quote/{ticker}/analysis"
            
            response = self.session.get(url, timeout=10)
            response.raise_for_status()
            
            if response.status_code == 200:
                return {
                    'ticker': ticker,
                    'buy_count': 'REQUIRES_PARSING',
                    'hold_count': 'REQUIRES_PARSING',
                    'sell_count': 'REQUIRES_PARSING',
                    'consensus_rating': 'REQUIRES_DATA_EXTRACTION',
                    'source': 'Yahoo Finance',
                    'source_tier': 2,
                    'confidence': 0.85
                }
            else:
                return {
                    'ticker': ticker,
                    'error': 'Failed to fetch analyst data',
                    'source': 'Yahoo Finance'
                }
        
        except Exception as e:
            return {
                'ticker': ticker,
                'error': str(e),
                'source': 'Yahoo Finance'
            }
    
    def get_price_targets(self, ticker: str) -> Dict:
        """
        Get analyst price targets
        - Average target, high, low, current price vs target
        
        Source: Yahoo Finance (Tier 2)
        Returns: {ticker, avg_target, high_target, low_target, current_price, upside_pct}
        """
        
        try:
            url = f"https://finance.yahoo.com/quote/{ticker}"
            
            response = self.session.get(url, timeout=10)
            response.raise_for_status()
            
            return {
                'ticker': ticker,
                'avg_target': 'REQUIRES_PARSING',
                'high_target': 'REQUIRES_PARSING',
                'low_target': 'REQUIRES_PARSING',
                'current_price': 'REQUIRES_PARSING',
                'upside_downside_pct': None,
                'source': 'Yahoo Finance',
                'source_tier': 2,
                'confidence': 0.85
            }
        
        except Exception as e:
            return {
                'ticker': ticker,
                'error': str(e),
                'source': 'Yahoo Finance'
            }
    
    def get_earnings_estimates(self, ticker: str) -> Dict:
        """
        Get analyst earnings estimates (EPS, Revenue)
        - Current quarter, next quarter, next fiscal year
        
        Returns: {ticker, current_eps_est, next_eps_est, revenue_growth, source}
        """
        
        try:
            url = f"https://finance.yahoo.com/quote/{ticker}/analysis"
            
            response = self.session.get(url, timeout=10)
            response.raise_for_status()
            
            return {
                'ticker': ticker,
                'current_quarter_eps': 'REQUIRES_PARSING',
                'next_quarter_eps': 'REQUIRES_PARSING',
                'fiscal_year_eps': 'REQUIRES_PARSING',
                'revenue_growth_estimate': 'REQUIRES_PARSING',
                'source': 'Yahoo Finance',
                'source_tier': 2,
                'confidence': 0.80
            }
        
        except Exception as e:
            return {
                'ticker': ticker,
                'error': str(e),
                'source': 'Yahoo Finance'
            }
    
    def get_rating_trends(self, ticker: str, days_back: int = 90) -> Dict:
        """
        Track analyst rating changes over time
        - Upgrades vs downgrades in last N days
        
        Returns: {ticker, upgrades, downgrades, net_change, trend}
        """
        
        return {
            'ticker': ticker,
            'upgrades': 'REQUIRES_DATA_AGGREGATION',
            'downgrades': 'REQUIRES_DATA_AGGREGATION',
            'net_change': None,
            'trend': 'REQUIRES_TIME_SERIES',
            'period_days': days_back,
            'source': 'Yahoo Finance / Seeking Alpha',
            'source_tier': 2
        }
    
    def calculate_consensus_signal(self,
                                   buy_pct: float = None,
                                   hold_pct: float = None,
                                   sell_pct: float = None,
                                   upside_pct: float = None,
                                   rating_trend: str = None) -> Dict:
        """
        Calculate consensus analyst signal (0-100)
        
        Factors:
        - Buy % > 50% = bullish (+30 pts)
        - Buy % > 70% = very bullish (+40 pts)
        - Sell % > 30% = bearish (-30 pts)
        - Upside > 15% = bullish (+15 pts)
        - Rating upgrades = bullish trend (+10 pts)
        
        Returns: {consensus_signal_score, components, recommendation}
        """
        
        signal_score = 50  # Neutral baseline
        components = {}
        
        # Buy/Hold/Sell distribution analysis
        if buy_pct:
            if buy_pct > 70:
                signal_score += 40
                components['buy_pct'] = 'VERY_BULLISH_CONSENSUS'
            elif buy_pct > 50:
                signal_score += 30
                components['buy_pct'] = 'BULLISH_CONSENSUS'
            else:
                signal_score -= 15
                components['buy_pct'] = 'MIXED_CONSENSUS'
        
        if sell_pct and sell_pct > 30:
            signal_score -= 30
            components['sell_pct'] = 'BEARISH_CONSENSUS'
        
        # Price target upside analysis
        if upside_pct:
            if upside_pct > 20:
                signal_score += 20
                components['upside'] = 'HIGH_UPSIDE'
            elif upside_pct > 10:
                signal_score += 10
                components['upside'] = 'MODERATE_UPSIDE'
            elif upside_pct < -10:
                signal_score -= 20
                components['upside'] = 'DOWNSIDE_RISK'
        
        # Rating trend
        if rating_trend == 'UPGRADING':
            signal_score += 15
            components['trend'] = 'POSITIVE_MOMENTUM'
        elif rating_trend == 'DOWNGRADING':
            signal_score -= 15
            components['trend'] = 'NEGATIVE_MOMENTUM'
        
        signal_score = max(0, min(100, signal_score))
        
        if signal_score > 65:
            recommendation = 'STRONG_BUY'
        elif signal_score > 55:
            recommendation = 'BUY'
        elif signal_score < 35:
            recommendation = 'SELL'
        elif signal_score < 45:
            recommendation = 'REDUCE'
        else:
            recommendation = 'HOLD'
        
        return {
            'consensus_signal_score': signal_score,
            'max_score': 100,
            'components': components,
            'recommendation': recommendation,
            'confidence': 0.82,
            'source': 'analyst_consensus.py'
        }

if __name__ == "__main__":
    analyst = AnalystConsensus()
    
    for ticker in ['CRWD', 'NET', 'RKLB', 'MP']:
        print(f"\n{ticker} - Analyst Consensus:")
        
        ratings = analyst.get_analyst_ratings(ticker)
        print(f"  Consensus: {ratings.get('consensus_rating', 'N/A')}")
        
        targets = analyst.get_price_targets(ticker)
        print(f"  Avg Target: {targets.get('avg_target', 'N/A')}")
        
        signal = analyst.calculate_consensus_signal(
            buy_pct=65,
            upside_pct=18,
            rating_trend='UPGRADING'
        )
        print(f"  Consensus Signal: {signal['consensus_signal_score']}/100 ({signal['recommendation']})")
