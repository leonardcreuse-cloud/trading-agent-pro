#!/usr/bin/env python3
"""
Scoring Signal - Combine all signals (v2: with reliability + confidence)
"""

from .price_technical import PriceTechnical
from .news_processor import NewsProcessor
from .scoring_fundamentals import ScoringFundamentals
from .scoring_reliability import ScoringReliability
from .scoring_confidence import ScoringConfidence
from datetime import datetime


class ScoringSignal:
    """Combine all signals into final recommendation (v2)"""
    
    def __init__(self):
        self.technical = PriceTechnical()
        self.news = NewsProcessor()
        self.fundamentals = ScoringFundamentals()
        self.reliability = ScoringReliability()
        self.confidence = ScoringConfidence()
    
    def combine_scores(self, tech_score, fund_score, reliability_score, confidence_score):
        """Weighted combination v2 (optimized weights)"""
        weights = {
            'technical': 0.25,
            'fundamentals': 0.55,
            'reliability': 0.10,
            'confidence': 0.10
        }
        
        combined = (tech_score * weights['technical'] +
                   fund_score * weights['fundamentals'] +
                   reliability_score * weights['reliability'] +
                   confidence_score * weights['confidence'])
        
        return round(combined, 2)
    
    def generate_final_signal(self, combined_score):
        """Convert combined score to BUY/SELL/HOLD"""
        if combined_score >= 68:
            return 'BUY'
        elif combined_score <= 35:
            return 'SELL'
        else:
            return 'HOLD'
    
    def analyze(self, ticker):
        """Complete signal analysis v2"""
        print(f"  [SIGNAL] {ticker}...")
        
        # Get all component scores
        tech_result = self.technical.analyze(ticker)
        fund_result = self.fundamentals.analyze(ticker)
        news_result = self.news.analyze(ticker)
        
        tech_score = tech_result.get('technical_score', 50)
        fund_score = fund_result.get('fundamental_score', 50)
        news_score = news_result.get('news_score', 50)
        
        # Get reliability
        reliability_result = self.reliability.analyze(
            ticker,
            tech_result.get('signal', 'HOLD'),
            fund_result.get('signal', 'HOLD'),
            news_result.get('signal', 'HOLD')
        )
        reliability_score = reliability_result.get('reliability_score', 50)
        
        # Get confidence
        confidence_result = self.confidence.analyze(ticker, reliability_score)
        confidence_score = confidence_result.get('confidence_score', 50)
        
        # Combine all
        combined = self.combine_scores(tech_score, fund_score, reliability_score, confidence_score)
        final_signal = self.generate_final_signal(combined)
        
        # Calculate agreement
        signals = [
            tech_result.get('signal', 'HOLD'),
            fund_result.get('signal', 'HOLD'),
            news_result.get('signal', 'HOLD')
        ]
        agreement = signals.count(final_signal)
        confidence_pct = (agreement / len(signals)) * 100
        
        return {
            'ticker': ticker,
            'signal': final_signal,
            'combined_score': combined,
            'confidence_pct': round(confidence_pct, 2),
            'component_scores': {
                'technical': tech_score,
                'fundamentals': fund_score,
                'reliability': reliability_score,
                'confidence': confidence_score
            },
            'component_signals': {
                'technical': tech_result.get('signal', 'HOLD'),
                'fundamentals': fund_result.get('signal', 'HOLD'),
                'news': news_result.get('signal', 'HOLD')
            },
            'recommendation': f"{final_signal} (Score: {combined})",
            'timestamp': datetime.now().isoformat(),
            'source': 'Multi-Source Real Data v2',
            'confidence': round(confidence_pct, 2)
        }


if __name__ == "__main__":
    signal = ScoringSignal()
    for ticker in ['CRWD', 'NET', 'RKLB', 'MP']:
        result = signal.analyze(ticker)
        print(f"    âœ“ {ticker}: {result['recommendation']}")
